	   FileNamePrefix = Sets$FilePrefix[Index]

	   TheseRaws = subset(DIAuditorByRun, startsWith(SourceFile, FileNamePrefix))
	   Sets$RawCount[[Index]] = length(TheseRaws$RTDuration)
	   Sets$RTDuration[[Index]] = median(TheseRaws$RTDuration)
	   Sets$MS1CycleTime[[Index]] = median(TheseRaws$MS1CycleTime)
	   Sets$MSnCycleTime[[Index]] = median(TheseRaws$AverageMedianCycleTime)
	   Sets$MSnRate[[Index]] = median(TheseRaws$mzMLMSnCount/(TheseRaws$RTDuration*60))
	   Sets$IsolationWindowCount[[Index]] = median(TheseRaws$IsolationWindowCount)
	   # Grab only the isolation windows that pertain to this particular instrument set
	   TheseRuns = subset(DIAuditorByIsolationWindow,startsWith(SourceFile, FileNamePrefix))
	   TheseRaws = unique(TheseRuns$SourceFile)
	   NumRaws = length(TheseRaws)
	   NumWindows = dim(TheseRuns)[1] / NumRaws
	   TIC50ileRTMatrix = matrix(nrow=NumRaws, ncol=NumWindows, 0)
	   rownames(TIC50ileRTMatrix) = TheseRaws
	   CurrentRow=1
	   for (ThisRaw in TheseRaws) {
	       ThisVector = subset(TheseRuns,SourceFile==ThisRaw)$TIC50ileRT
	       TIC50ileRTMatrix[CurrentRow,] = ThisVector
	       CurrentRow = CurrentRow+1
	       }
	   MDist <- dist(TIC50ileRTMatrix)
	   Sets$MinDistance[[Index]] = min(MDist)
	   Sets$MedianDistance[[Index]] = median(MDist)
	   Sets$MaxDistance[[Index]] = max(MDist)
}

SpectronautStats <- function() {
	   PG <- read.table("Spectronaut_23\\20260801_134323_xevo_Report.tsv",header=T,sep="\t",quote="")
	   SampleAs <- grep("_SA_",names(PG))
	   SampleBs <- grep("_SB_",names(PG))
	   SampleCs <- grep("_SC_",names(PG))
	   BGAL <- PG[grep("P00722",PG$PG.ProteinGroups),]
	   LYSC <- PG[grep("P00698",PG$PG.ProteinGroups),]
	   AMYG <- PG[grep("P69328",PG$PG.ProteinGroups),]
	   PROG <- PG[grep("Q54181",PG$PG.ProteinGroups),]
	   Intensities <- data.matrix(PG[,2:ncol(PG)])
	   row.names(Intensities) <- PG$PG.ProteinGroups
	   AllIntensities <- Intensities[complete.cases(Intensities),]
	   means <- rowMeans(AllIntensities)
	   sds <- apply(AllIntensities, 1, sd)
	   TotalIntensity <- rowSums(AllIntensities)
	   CVs <- sds/means
	   Variation <- data.frame(TotalIntensity,CVs)
	   TopHalf <- subset(Variation,TotalIntensity > median(TotalIntensity))
	   BottomHalf <- subset(Variation,TotalIntensity < median(TotalIntensity))
	   # Compute all log fold changes between samples B and samples A
	   BGALfc <- outer(log2(na.exclude(as.numeric(BGAL[SampleBs]))),log2(na.exclude(as.numeric(BGAL[SampleAs]))),FUN="-")
	   LYSCfc <- outer(log2(na.exclude(as.numeric(LYSC[SampleBs]))),log2(na.exclude(as.numeric(LYSC[SampleAs]))),FUN="-")
	   AMYGfc <- outer(log2(na.exclude(as.numeric(AMYG[SampleBs]))),log2(na.exclude(as.numeric(AMYG[SampleAs]))),FUN="-")
	   PROGfc <- outer(log2(na.exclude(as.numeric(PROG[SampleBs]))),log2(na.exclude(as.numeric(PROG[SampleAs]))),FUN="-")
	   
	   #Compute metrics explicitly
	   ExptCount = dim(Intensities)[2]
	   ### Manually grabbed the two following values from Spectronaut Log
	   ### grep precursors xevo_AnalysisLog.txt; Text to Columns in Excel
	   IDPrecursorsMedian = 21980
	   IDProteinsMedian = 3403
	   ### Manually grabbed the two following values from Spectronaut Run Summaries
	   ### grep MS1.Average.Delta *.tsv; Text to Columns in Excel
	   MassAccuracyMS1Median = 6.69
	   ### grep MS2.Average.Delta *.tsv; Text to Columns in Excel
	   MassAccuracyMS2Median = 8.39
	   QuantProteinsAny = dim(PG)[1]
	   QuantProteinsAll = dim(AllIntensities)[1]
	   CVTopHalfMedian = median(TopHalf$CVs)
	   CVBottomHalfMedian = median(BottomHalf$CVs)
	   BGALfcMedian = median(BGALfc)
	   LYSCfcMedian = median(LYSCfc)
	   AMYGfcMedian = median(AMYGfc)
	   PROGfcMedian = median(PROGfc)
	   BGALFalsePos = sum(!is.na(BGAL[SampleCs]))
	   LYSCFalsePos = sum(!is.na(LYSC[SampleCs]))
	   AMYGFalsePos = sum(!is.na(AMYG[SampleCs]))
	   PROGFalsePos = sum(!is.na(PROG[SampleCs]))
	   BGALFalseNeg = length(SampleBs) - sum(BGAL[SampleBs]>0)
	   LYSCFalseNeg = length(SampleBs) - sum(LYSC[SampleBs]>0)
	   AMYGFalseNeg = length(SampleBs) - sum(AMYG[SampleBs]>0)
	   PROGFalseNeg = length(SampleBs) - sum(PROG[SampleBs]>0)
	   directory = "Spectronaut_23"
	   Metrics <- data.frame(directory,ExptCount,IDPrecursorsMedian,IDProteinsMedian,MassAccuracyMS1Median,MassAccuracyMS2Median,QuantProteinsAny,QuantProteinsAll,
	   	   CVTopHalfMedian,CVBottomHalfMedian,BGALfcMedian,LYSCfcMedian,AMYGfcMedian,PROGfcMedian,
		   BGALFalsePos,LYSCFalsePos,AMYGFalsePos,PROGFalsePos,
		   BGALFalseNeg,LYSCFalseNeg,AMYGFalseNeg,PROGFalseNeg)
	   return(Metrics)
	   }

# Create the empty vectors we will be populating below
Sets$RTDuration = 0
Sets$MS1CycleTime = 0
Sets$MSnCycleTime = 0
Sets$MSnRate = 0
Sets$IsolationWindowCount = 0
Sets$MedianIsolationWindowWidth = 0
Sets$MinDistance = 0
Sets$MedianDistance = 0
Sets$MaxDistance = 0

# Derive Instrument-Level Stats
for (Index in 1:length(Sets$FilePrefix)) {
	   FileNamePrefix = Sets$FilePrefix[Index]
	   # Pull out the ByRun metrics rows that pertain to this particular instrument set
	   TheseRaws = subset(DIAuditorByRun, startsWith(SourceFile, FileNamePrefix))
	   Sets$RTDuration[[Index]] = median(TheseRaws$RTDuration)
	   Sets$MS1CycleTime[[Index]] = median(TheseRaws$MS1CycleTime)
	   Sets$MSnCycleTime[[Index]] = median(TheseRaws$AverageMedianCycleTime)
	   Sets$MSnRate[[Index]] = median(TheseRaws$mzMLMSnCount/(TheseRaws$RTDuration*60))
	   Sets$IsolationWindowCount[[Index]] = median(TheseRaws$IsolationWindowCount)
	   # Grab only the isolation windows that pertain to this particular instrument set
	   TheseRuns = subset(DIAuditorByIsolationWindow,startsWith(SourceFile, FileNamePrefix))
	   TheseRaws = unique(TheseRuns$SourceFile)
	   NumRaws = length(TheseRaws)
	   NumWindows = dim(TheseRuns)[1] / NumRaws
	   Sets$MedianIsolationWindowWidth[[Index]] = median(TheseRuns$WidthMZ)
	   cat(FileNamePrefix,": ",NumRaws,"\t",NumWindows,"\n");
	   if (NumRaws > 1) {
	   TIC50ileRTMatrix = matrix(nrow=NumRaws, ncol=NumWindows, 0)
	   rownames(TIC50ileRTMatrix) = TheseRaws
	   CurrentRow=1
	   for (ThisRaw in TheseRaws) {
	       ThisVector = subset(TheseRuns,SourceFile==ThisRaw)$TIC50ileRT
	       TIC50ileRTMatrix[CurrentRow,] = ThisVector
	       CurrentRow = CurrentRow+1
	       }
	   MDist <- dist(TIC50ileRTMatrix)
	   Sets$MinDistance[[Index]] = min(MDist)
	   Sets$MedianDistance[[Index]] = median(MDist)
	   Sets$MaxDistance[[Index]] = max(MDist)
	   }
}
